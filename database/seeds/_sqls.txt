-- вывод предшествующего сальдо вместе с текущим сальдо

select d.*, orgSaldo_onDate(d.orgid,d.ownorgid,d.prev_operdate) as prev_saldo, orgSaldo_onDate(d.orgid,d.ownorgid, null) as cur_saldo
from
 (with x as (select distinct orgid, ownorgid from paydocs)
	select x.ownorgid,x.orgid
    , ( select a.operdate from (
        SELECT DISTINCT mr.wrkdate as operdate, mro.orgid, mro.suporgid as ownorgid
        from mr_opers as mro
        join mchn_raids as mr on mr.id=mro.mr_id
        where mro.orgid=x.orgid and mro.suporgid=x.ownorgid

        union
        select paydate as operdate, orgid, ownorgid
        from paydocs as pd
        where pd.orgid=x.orgid and pd.ownorgid=x.ownorgid
        order by operdate desc limit 2,1
        ) a) prev_operdate

    from x
) d
where d.prev_operdate is not null
order by cur_saldo

Получить даты месяца

select FROM_UNIXTIME(UNIX_TIMESTAMP(CONCAT('2022-02-',n)),'%Y-%m-%d') as Date
	from (
        select (((b4.0 << 1 | b3.0) << 1 | b2.0) << 1 | b1.0) << 1 | b0.0 as n
                from  (select 0 union all select 1) as b0,
                      (select 0 union all select 1) as b1,
                      (select 0 union all select 1) as b2,
                      (select 0 union all select 1) as b3,
                      (select 0 union all select 1) as b4 ) t
        where n > 0 and n <= day(last_day('2022-02-01')) order by date
