    and dw.wrkdate>='2025-05-01'

	set dw.night_hr_rate = hr_night_Rate(dw.wrkdate, dw.staffid, dw.wrktypeid)

    and dw.wrkdate>='2025-05-01'

    and dw.wrkdate>='2025-05-01'
    and dw.hrs_salary != dw.day_wrkhrs * dw.day_hr_rate + dw.night_wrkhrs * dw.night_hr_rate;

CREATE DEFINER=`sns25`@`localhost` FUNCTION `hr_day_rate`( p_wrkdate date, p_staffid bigint, p_wrktypeid bigint) RETURNS decimal(10,0)
    READS SQL DATA
BEGIN
	declare rslt decimal;

    select i.hr_day_rate into rslt
		from srs_hr_items as i
		join salary_rate_sets as srs on srs.id = i.srs_id
		join orgstaff as os on os.id = p_staffid
		where i.wrktypeid = p_wrktypeid
			and srs.payrolltypeid=1
			and ifnull(srs.ownorgid,os.orgid)=os.orgid
			and p_wrkdate between srs.begdate and ifnull(srs.enddate, p_wrkdate)
			and TIMESTAMPDIFF(year, ifnull(os.begdate, p_wrkdate), p_wrkdate ) between i.min_wrkexp and i.max_wrkexp-0.001
			;
RETURN rslt;
END


CREATE DEFINER=`homestead`@`%` FUNCTION `hr_night_rate`( p_wrkdate date, p_staffid bigint, p_wrktypeid bigint) RETURNS decimal(10,0)
    READS SQL DATA
BEGIN
	declare rslt decimal;

    select i.hr_night_rate into rslt
		from srs_hr_items as i
		join salary_rate_sets as srs on srs.id = i.srs_id
		join orgstaff as os on os.id = p_staffid
		where i.wrktypeid = p_wrktypeid
			and srs.payrolltypeid=1
			and ifnull(srs.ownorgid,os.orgid)=os.orgid
			and p_wrkdate between srs.begdate and ifnull(srs.enddate, p_wrkdate)
			and TIMESTAMPDIFF(year, ifnull(os.begdate, p_wrkdate), p_wrkdate ) between i.min_wrkexp and i.max_wrkexp-0.001
			;

RETURN rslt;
END

select d.*, orgSaldo_onDate(d.orgid,d.ownorgid,d.prev_operdate) as prev_saldo, orgSaldo_onDate(d.orgid,d.ownorgid, null) as cur_saldo
from
 (with x as (select distinct orgid, ownorgid from paydocs)
	select x.ownorgid,x.orgid
    , ( select a.operdate from (
        SELECT DISTINCT mr.wrkdate as operdate, mro.orgid, mro.suporgid as ownorgid

        join mchn_raids as mr on mr.id=mro.mr_id

        union
        select paydate as operdate, orgid, ownorgid
        from paydocs as pd
        where pd.orgid=x.orgid and pd.ownorgid=x.ownorgid
        order by operdate desc limit 2,1
        ) a) prev_operdate

    from x
) d
where d.prev_operdate is not null
order by cur_saldo

Получить даты месяца

select FROM_UNIXTIME(UNIX_TIMESTAMP(CONCAT('2022-02-',n)),'%Y-%m-%d') as Date
	from (
        select (((b4.0 << 1 | b3.0) << 1 | b2.0) << 1 | b1.0) << 1 | b0.0 as n
                from  (select 0 union all select 1) as b0,
                      (select 0 union all select 1) as b1,
                      (select 0 union all select 1) as b2,
                      (select 0 union all select 1) as b3,
                      (select 0 union all select 1) as b4 ) t
        where n > 0 and n <= day(last_day('2022-02-01')) order by date
